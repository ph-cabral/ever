SELECT CodCliente, Nombre, AnioMes,
       SUM(Cant)  AS CantidadNeta,
       SUM(Monto) AS MontoNeto
FROM (
SELECT
    c.CodCliente AS CodCliente,
    LTRIM(RTRIM(c.Cliente_Nombre)) AS Nombre,
    CASE WHEN vc.FecMovim BETWEEN 81819 AND 81849 THEN 202501 WHEN vc.FecMovim BETWEEN 81850 AND 81877 THEN 202502 WHEN vc.FecMovim BETWEEN 81878 AND 81908 THEN 202503 WHEN vc.FecMovim BETWEEN 81909 AND 81938 THEN 202504 WHEN vc.FecMovim BETWEEN 81939 AND 81969 THEN 202505 WHEN vc.FecMovim BETWEEN 81970 AND 81999 THEN 202506 WHEN vc.FecMovim BETWEEN 82000 AND 82030 THEN 202507 WHEN vc.FecMovim BETWEEN 82031 AND 82061 THEN 202508 WHEN vc.FecMovim BETWEEN 82062 AND 82091 THEN 202509 WHEN vc.FecMovim BETWEEN 82092 AND 82122 THEN 202510 WHEN vc.FecMovim BETWEEN 82123 AND 82152 THEN 202511 WHEN vc.FecMovim BETWEEN 82153 AND 82183 THEN 202512 WHEN vc.FecMovim BETWEEN 82184 AND 82214 THEN 202601 WHEN vc.FecMovim BETWEEN 82215 AND 82242 THEN 202602 WHEN vc.FecMovim BETWEEN 82243 AND 82273 THEN 202603 WHEN vc.FecMovim BETWEEN 82274 AND 82303 THEN 202604 WHEN vc.FecMovim BETWEEN 82304 AND 82334 THEN 202605 WHEN vc.FecMovim BETWEEN 82335 AND 82364 THEN 202606 WHEN vc.FecMovim BETWEEN 82365 AND 82395 THEN 202607 WHEN vc.FecMovim BETWEEN 82396 AND 82426 THEN 202608 WHEN vc.FecMovim BETWEEN 82427 AND 82456 THEN 202609 WHEN vc.FecMovim BETWEEN 82457 AND 82487 THEN 202610 WHEN vc.FecMovim BETWEEN 82488 AND 82517 THEN 202611 WHEN vc.FecMovim BETWEEN 82518 AND 82548 THEN 202612 ELSE NULL END AS AnioMes,
    CASE cc.DebitoCredito WHEN 1 THEN r.Cantidad ELSE r.Cantidad * -1 END AS Cant,
    CASE cc.DebitoCredito WHEN 1 THEN (r.Cantidad * r.PrecioVenta) ELSE (r.Cantidad * r.PrecioVenta) * -1 END AS Monto
FROM MAGNUS_SITD.dbo.Clientes c
JOIN Ven_CompCabecera vc ON vc.CodCliente = c.CodCliente
JOIN Ven_CompRenglon r   ON r.NroMovVenta = vc.NroMovVenta

LEFT JOIN StkFer_Articulos  s  ON s.CodArticulo    = r.CodArticu
LEFT JOIN StkFer_ArtParamet ap ON ap.ArticuloPatron = s.ArticuloPatron
WHERE cc.EvitaInformesYListados <> 1
  AND vc.FecMovim BETWEEN ? AND ?
  AND ap.Nivel1 IN (SELECT n.Nivel1 FROM Stk_Nivel1 n WHERE LTRIM(RTRIM(n.Detalle)) = ?)
) t
WHERE AnioMes IS NOT NULL
GROUP BY CodCliente, Nombre, AnioMes
